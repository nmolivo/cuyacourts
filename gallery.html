Caseload Per Annum

SELECT '20' || SPLIT_PART(case_number, '-', 2) as start_year, judge_name, count(DISTINCT SPLIT_PART(case_number, '-', 3)) as count
FROM case_defendant 
WHERE status LIKE '%CASE CLOSED%'
GROUP BY start_year, judge_name
